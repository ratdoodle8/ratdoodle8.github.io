function getUsername() { const params = new URLSearchParams( window.location.search ); const profile = params.get("profile"); if (profile) { return profile.trim(); } const parts = window.location.pathname .split("/") .filter(Boolean); if ( parts.length >= 2 && parts[0] === "user" ) { return decodeURIComponent( parts[1] ).trim(); } return ""; } const username = getUsername(); if ( new URLSearchParams( window.location.search ).has("profile") && username ) { history.replaceState( null, "", "/user/" + encodeURIComponent(username) ); }
